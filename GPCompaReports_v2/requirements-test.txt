# Browser smoke tests only (tests/test_smoke_browser.py).
#
# Deliberately separate from the site's own dependencies. Nothing here is
# needed to build or serve the site, and the ordinary test suite skips these
# tests when Playwright is absent.
#
# The system Python is externally managed (PEP 668), so install into a venv:
#
#     python3 -m venv .venv-test
#     .venv-test/bin/python -m pip install -r GPCompaReports_v2/requirements-test.txt
#     .venv-test/bin/python -m playwright install chromium
#
# `playwright install chromium` downloads a private Chromium (~400 MB) into
# ~/.cache/ms-playwright. It is not the snap Chromium and is not confined by
# snap, which is deliberate: snap's `home` interface silently denies access to
# hidden directories, and that class of trap has cost this project time before.
playwright==1.62.0
pytest>=8
